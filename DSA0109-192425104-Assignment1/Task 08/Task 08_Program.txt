
#include <iostream>
#include <iomanip>
#include <limits>
using namespace std;

int main()
{
    long long balance, fare;

    cout << "Enter balance in paise: ";

    while (!(cin >> balance) || balance < 0)
    {
        cout << "Invalid balance. Enter again: ";
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
    }

    cout << "Enter fare in paise: ";

    while (!(cin >> fare) || fare < 0)
    {
        cout << "Invalid fare. Enter again: ";
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
    }

    if (balance - fare >= 5000)
    {
        balance = balance - fare;

        cout << fixed << setprecision(2);
        cout << "Journey approved\n";
        cout << "Remaining balance: Rs "
             << balance / 100.0 << '\n';
    }
    else
    {
        cout << "Journey denied\n";
        cout << "Balance unchanged: Rs "
             << balance / 100.0 << '\n';
    }

    return 0;
}
