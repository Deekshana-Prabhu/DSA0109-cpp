
#include <iostream>
#include <climits>
using namespace std;

int main() {
    int fare = 3000;
    long long journeys = 3000000;
    long long totalFare;

    float a = 0.1f;
    float b = 0.2f;

    totalFare = journeys * fare;

    cout << "Size of int: "
         << sizeof(int) << " bytes" << endl;

    cout << "Size of long long: "
         << sizeof(long long) << " bytes" << endl;

    cout << "Maximum int value: "
         << INT_MAX << endl;

    cout << "Total fare in paise: "
         << totalFare << endl;

    cout << "Float sum: "
         << a + b << endl;

    return 0;
}
