#include <iostream>
#include <iomanip>
using namespace std;

enum Zone
{
    Inner,
    Middle,
    Outer,
    Airport
};

const int fare[] = {1000, 2000, 3000, 6000};

class SmartCard
{
private:
    long long balance;

public:
    SmartCard(long long b)
    {
        balance = b;
    }

    void travel(Zone z, bool peak, bool weekend)
    {
        long long cost = fare[z];

        if (weekend)
            cost = cost - cost * 10 / 100;
        else if (peak)
            cost = cost + cost * 20 / 100;

        if (balance - cost >= 5000)
        {
            balance = balance - cost;

            cout << "Journey approved" << endl;
            cout << "Fare: Rs. "
                 << cost / 100.0 << endl;
        }
        else
        {
            cout << "Journey denied: Low balance"
                 << endl;
        }

        cout << "Balance: Rs. "
             << balance / 100.0 << endl;
    }
};

class MonthlyPass
{
private:
    Zone zone1, zone2;

public:
    MonthlyPass(Zone z1, Zone z2)
    {
        zone1 = z1;
        zone2 = z2;
    }

    bool isValid(Zone z)
    {
        return (z == zone1 || z == zone2);
    }
};

int main()
{
    cout << fixed << setprecision(2);

    SmartCard card(20000);

    cout << "Weekend Airport Journey" << endl;
    card.travel(Airport, false, true);

    MonthlyPass pass(Inner, Middle);

    cout << "\nMonthly Pass Test" << endl;

    if (pass.isValid(Inner))
        cout << "Pass valid for Inner zone" << endl;
    else
        cout << "Pass not valid" << endl;

    if (pass.isValid(Outer))
        cout << "Pass valid for Outer zone" << endl;
    else
        cout << "Pass not valid for Outer zone" << endl;

    return 0;
}
